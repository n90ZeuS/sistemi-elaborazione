Funzione analizza_file

from pathlib import Path


def analizza_file(percorso: Path) -> dict[str, int]:
    """Legge un file di testo e restituisce statistiche di base."""
    num_righe: int = 0
    num_righe_non_vuote: int = 0
    num_parole: int = 0
    num_caratteri: int = 0

    with open(percorso, "r", encoding="utf-8") as f:
        for riga in f:
            num_righe += 1
            num_caratteri += len(riga)
            if riga.strip():
                num_righe_non_vuote += 1
                num_parole += len(riga.split())

    return {
        "righe": num_righe,
        "righe_non_vuote": num_righe_non_vuote,
        "parole": num_parole,
        "caratteri": num_caratteri,
    }

from pathlib import Path


def salva_statistiche(
    percorso_output: Path,
    nome_sorgente: str,
    statistiche: dict[str, int],
) -> None:
    """Salva le statistiche di un file in un report testuale."""
    # Creiamo la directory genitore se non esiste
    percorso_output.parent.mkdir(parents=True, exist_ok=True)

    with open(percorso_output, "w", encoding="utf-8") as f:
        f.write(f"=== Report per: {nome_sorgente} ===\n\n")
        for chiave, valore in statistiche.items():
            f.write(f"  {chiave:>20}: {valore}\n")
        f.write(f"\n{'='*40}\n")

    print(f"Report salvato in: {percorso_output}")

Caricare un CSV con DictReader

import csv
from pathlib import Path


def carica_studenti(percorso: Path) -> list[dict[str, str]]:
    """Carica un CSV di studenti con separatore punto e virgola."""
    studenti: list[dict[str, str]] = []

    with open(percorso, "r", encoding="utf-8") as f:
        lettore = csv.DictReader(f, delimiter=";")
        for riga in lettore:
            studenti.append(dict(riga))

    return studenti


def calcola_media_voti(studenti: list[dict[str, str]]) -> float:
    """Calcola la media dei voti degli studenti."""
    if not studenti:
        raise ValueError("Impossibile calcolare la media: lista vuota")

    voti: list[int] = [int(s["voto"]) for s in studenti]
    return sum(voti) / len(voti)

Caricare e navigare JSON

import json
from pathlib import Path


def carica_corso(percorso: Path) -> dict:
    """Carica i dati di un corso da file JSON."""
    with open(percorso, "r", encoding="utf-8") as f:
        dati: dict = json.load(f)
    return dati


def riepilogo_corso(dati: dict) -> None:
    """Stampa un riepilogo del corso con le medie di ogni studente."""
    print(f"Corso: {dati['nome_corso']}")
    print(f"Docente: {dati['docente']['nome']} {dati['docente']['cognome']}")

    for s in dati["studenti"]:
        voti: list[int] = s["voti"]
        if voti:
            media: float = sum(voti) / len(voti)
            print(f"  {s['matricola']} {s['nome']:<12} media: {media:.2f}")
        else:
            print(f"  {s['matricola']} {s['nome']:<12} media: N/D")

Salvare risultati in JSON

def salva_medie_json(dati: dict, percorso_output: Path) -> None:
    """Calcola le medie e le salva in un nuovo file JSON."""
    risultati: list[dict[str, str | float | None]] = []

    for s in dati["studenti"]:
        voti: list[int] = s["voti"]
        media: float | None = sum(voti) / len(voti) if voti else None
        risultati.append({
            "matricola": s["matricola"],
            "nome": s["nome"],
            "media": round(media, 2) if media is not None else None,
        })

    output: dict[str, str | list] = {
        "corso": dati["nome_corso"],
        "medie": risultati,
    }

    percorso_output.parent.mkdir(parents=True, exist_ok=True)
    with open(percorso_output, "w", encoding="utf-8") as f:
        json.dump(output, f, indent=2, ensure_ascii=False)
indent=2 → JSON leggibile
ensure_ascii=False → preserva gli accenti

json.load vs json.loads

json.load(f)

Legge da file

with open("dati.json") as f:
    dati = json.load(f)

json.loads(s)

Legge da stringa

testo = '{"chiave": 42}'
dati = json.loads(testo)
Nota: la s finale sta per string. Lo stesso vale per json.dump (su file) e json.dumps (restituisce stringa).

Pattern fondamentale: eccezioni specifiche

Eccezioni specifiche

try:
    valore = float(testo)
except ValueError:
    valore = None

try:
    with open(path) as f:
        dati = f.read()
except FileNotFoundError:
    print("File non trovato")
except UnicodeDecodeError:
    print("Encoding errato")

Eccezione generica

try:
    valore = float(testo)
except Exception:
    # Nasconde TUTTI i bug!
    valore = None




# Non fatelo mai:
except:
    pass
Regola: catturate solo le eccezioni che sapete gestire. Un except Exception generico nasconde i bug.

Esercizi autonomi

Dal semplice al complesso

Esercizi base

Esercizio 1 — Analisi di un file di log Base

Scrivere analizza_log(percorso: Path) -> dict[str, int] che:

  1. Legge il file riga per riga
  2. Conta le righe per livello (INFO, WARNING, ERROR)
  3. Gestisce righe vuote o malformate senza interrompersi

Poi scrivere salva_errori(percorso_log, percorso_output) -> int che estrae solo le righe ERROR.

Esercizio 2 — Pulizia di un CSV sporco Base

Scrivere pulisci_csv(percorso_input, percorso_output) -> dict[str, int] che:

  1. Scarta righe senza prodotto
  2. Converte campi numerici, sostituendo valori invalidi con None
  3. Scrive un CSV pulito con solo le righe valide

Esercizi intermedi

Esercizio 3 — Conversione CSV / JSON Intermedio

Scrivere csv_a_json(percorso_csv, percorso_json) -> int che:

  • Rileva automaticamente il separatore (, o ;)
  • Converte valori numerici nei tipi Python corrispondenti
  • Salva come JSON con indent=2

Scrivere anche la funzione inversa json_a_csv.

Esercizio 4 — Unione di piu' CSV Intermedio

Scrivere unisci_e_riepiloga(cartella, output_unito, output_riepilogo) che:

  • Trova tutti i .csv con glob
  • Li unisce in un unico CSV (una sola intestazione)

Esercizio avanzato

Esercizio 5 — Analisi incrociata Avanzato

Due file di input: dipendenti.json e presenze.csv.

Scrivere report_presenze(path_dipendenti, path_presenze, path_output) che:

  1. Carica entrambi i file
  2. Per ogni dipendente, calcola ore totali e media giornaliera
  3. Raggruppa i risultati per reparto

Gestire: dipendenti senza presenze, presenze con ID inesistente, ore non numeriche.

Sfida finale — Le 6 fasi

CSV sporco → Caricamento → Pulizia → Statistiche → Output CSV + JSON + Report
  1. Caricamento: leggere il CSV gestendo encoding e separatore
  2. Pulizia: scartare righe senza stazione, convertire campi numerici, sostituire valori invalidi con None
  3. Statistiche per stazione: media, minimo, massimo per ogni campo numerico
  4. Output CSV: salvare in output/statistiche_meteo.csv
  5. Output JSON: salvare in output/statistiche_meteo.json
  6. Report testuale: riepilogo formattato a video
Ogni funzione con type hints completi. Il programma non deve mai crashare.
