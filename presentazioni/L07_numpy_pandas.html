Parte 1 — NumPy

Calcolo numerico vettorizzato

Creare array NumPy

L'ndarray e un contenitore di dati omogenei: operazioni su interi vettori senza cicli.

import numpy as np

# Da lista Python
voti: np.ndarray = np.array([28, 25, 30, 22, 27, 18, 30, 24])
print(f"Dtype: {voti.dtype}")    # int64
print(f"Shape: {voti.shape}")    # (8,)
print(f"Size:  {voti.size}")     # 8

# Funzioni di creazione
zeri: np.ndarray = np.zeros(10)
sequenza: np.ndarray = np.arange(0, 100, 5)      # da 0 a 95, passo 5
lineare: np.ndarray = np.linspace(0, 1, 11)       # 11 punti tra 0 e 1
Concetto chiave: un array NumPy ha un tipo unico (dtype) e dimensione fissa — questo lo rende molto piu efficiente di una lista Python.

Operazioni vettorizzate

Le operazioni aritmetiche si applicano elemento per elemento:

voti: np.ndarray = np.array([28, 25, 30, 22, 27, 18, 30, 24])

# Aritmetica elemento per elemento
voti_normalizzati: np.ndarray = voti / 30.0
voti_bonus: np.ndarray = voti + 2

# Operazioni tra array
pesi: np.ndarray = np.array([0.1, 0.1, 0.2, 0.1, 0.15, 0.1, 0.15, 0.1])
media_pesata: float = float(np.sum(voti * pesi))
print(f"Media pesata: {media_pesata:.2f}")

Confronti vettorizzati

# Restituiscono array di bool
sufficienti = voti >= 18
eccellenti = voti >= 28
print(np.sum(eccellenti))  # 3

Indicizzazione booleana

# Filtro diretto
voti_alti = voti[voti >= 27]
# array([28, 30, 27, 30])

Statistiche con NumPy

voti: np.ndarray = np.array([28, 25, 30, 22, 27, 18, 30, 24, 26, 29])

print(f"Media:               {np.mean(voti):.2f}")
print(f"Mediana:             {np.median(voti):.2f}")
print(f"Deviazione standard: {np.std(voti):.2f}")
print(f"Varianza:            {np.var(voti):.2f}")
print(f"Minimo:              {np.min(voti)}")
print(f"Massimo:             {np.max(voti)}")
print(f"Percentile 25:       {np.percentile(voti, 25):.2f}")
print(f"Percentile 75:       {np.percentile(voti, 75):.2f}")
Nota: queste funzioni sono ottimizzate in C e operano su blocchi contigui di memoria, sfruttando le istruzioni SIMD del processore.

Confronto prestazioni: Python vs NumPy

import numpy as np
import time

n: int = 1_000_000
dati_np: np.ndarray = np.random.randn(n)
dati_lista: list[float] = dati_np.tolist()

# Calcolo media con ciclo Python
inizio: float = time.time()
somma: float = 0.0
for valore in dati_lista:
    somma += valore
media_ciclo: float = somma / n
tempo_ciclo: float = time.time() - inizio

# Calcolo media con NumPy
inizio = time.time()
media_numpy: float = float(np.mean(dati_np))
tempo_numpy: float = time.time() - inizio

print(f"Ciclo Python: {tempo_ciclo:.4f} s")
print(f"NumPy:        {tempo_numpy:.6f} s")
print(f"NumPy e circa {tempo_ciclo / tempo_numpy:.0f}x piu veloce")
Risultato tipico: NumPy e 50–100x piu veloce. Il vantaggio cresce con la dimensione dei dati.

NumPy random: generazione di numeri casuali

import numpy as np

# Generatore con seed per riproducibilita
rng: np.random.Generator = np.random.default_rng(seed=42)

# Distribuzione uniforme
uniforme: np.ndarray = rng.uniform(low=0.0, high=1.0, size=10)

# Distribuzione normale
normale: np.ndarray = rng.normal(loc=25.0, scale=3.0, size=10)

# Distribuzione di Poisson
poisson: np.ndarray = rng.poisson(lam=5, size=10)

# Campione grande: verificare i parametri
campione: np.ndarray = rng.normal(loc=170, scale=10, size=10_000)
print(f"Media campionaria:    {np.mean(campione):.2f}")   # ~170
print(f"Dev. std campionaria: {np.std(campione):.2f}")    # ~10
Buona pratica: usate sempre default_rng(seed=...) per garantire la riproducibilita dei risultati.

Simulazione Monte Carlo: stimare pi greco

Generiamo punti casuali nel quadrato [-1,1] x [-1,1] e contiamo quanti cadono nel cerchio unitario.

def stima_pi_montecarlo(n_punti: int,
                        seed: int = 42) -> float:
    """Stima pi greco con il metodo Monte Carlo."""
    rng = np.random.default_rng(seed=seed)
    x = rng.uniform(-1, 1, size=n_punti)
    y = rng.uniform(-1, 1, size=n_punti)
    distanza_quadrata = x**2 + y**2
    dentro = int(np.sum(distanza_quadrata <= 1.0))
    return 4.0 * dentro / n_punti

for n in [100, 1_000, 10_000, 1_000_000]:
    stima = stima_pi_montecarlo(n)
    errore = abs(stima - np.pi)
    print(f"n={n:>10,}  pi~={stima:.5f}"
          f"  err={errore:.5f}")
Logica:
Area cerchio / Area quadrato = pi/4

Quindi: pi = 4 × (punti dentro / punti totali)

L'errore decresce come 1/√n

Parte 1 — Pandas

Manipolazione di dati tabulari

Creare un DataFrame e salvarlo come CSV

import pandas as pd
import numpy as np

rng: np.random.Generator = np.random.default_rng(seed=42)
n: int = 200

df: pd.DataFrame = pd.DataFrame({
    "id_studente": range(1, n + 1),
    "nome": [f"Studente_{i}" for i in range(1, n + 1)],
    "corso": rng.choice(
        ["Statistica", "Economia", "Sociologia", "Informatica"], size=n),
    "anno": rng.choice([1, 2, 3], size=n),
    "voto_medio": np.round(rng.normal(25, 3, size=n).clip(18, 30), 1),
    "crediti": rng.choice([30, 60, 90, 120, 150, 180], size=n),
    "regione": rng.choice(
        ["Lombardia", "Lazio", "Campania", "Sicilia",
         "Veneto", "Emilia-Romagna", "Toscana"], size=n),
})

df.to_csv("studenti.csv", index=False)
print(f"Dimensioni: {df.shape[0]} righe x {df.shape[1]} colonne")

Pulizia dei dati

# 1. Rimuovere duplicati
df_pulito: pd.DataFrame = df.drop_duplicates()

# 2. Correggere tipi sbagliati (errori -> NaN)
df_pulito["eta"] = pd.to_numeric(df_pulito["eta"], errors="coerce")

# 3. Normalizzare stringhe
df_pulito["corso"] = df_pulito["corso"].str.strip().str.capitalize()

# 4. Gestire i mancanti
# Rimuovere righe senza nome
df_pulito = df_pulito.dropna(subset=["nome"])
# Riempire voti mancanti con la mediana
mediana_voto: float = df_pulito["voto"].median()
df_pulito["voto"] = df_pulito["voto"].fillna(mediana_voto)
# Riempire eta mancante con la media
media_eta: float = df_pulito["eta"].mean()
df_pulito["eta"] = df_pulito["eta"].fillna(round(media_eta))
Attenzione: le decisioni prese nella pulizia (es. mediana vs eliminazione) possono influenzare significativamente i risultati dell'analisi.

dropna() vs fillna()

dropna()

Rimuove le righe (o colonne) con valori mancanti.

# Rimuovi righe con qualsiasi NaN
df.dropna()

# Solo dove manca il nome
df.dropna(subset=["nome"])

Quando: pochi mancanti, oppure il dato e essenziale (es. chiave primaria).

fillna()

Sostituisce i valori mancanti con un valore scelto.

# Con la mediana
df["voto"].fillna(df["voto"].median())

# Con un valore fisso
df["nome"].fillna("Sconosciuto")

Quando: molti mancanti, non si vuole perdere dati.

groupby

Il pattern split-apply-combine

groupby: statistiche per categoria

DataFrame → split (dividi per gruppo) → apply (calcola funzione) → combine (ricombina risultati) 
df: pd.DataFrame = pd.read_csv("studenti.csv")

# Media voto per corso
media_per_corso: pd.DataFrame = (
    df.groupby("corso")["voto_medio"]
    .mean()
    .round(2)
    .sort_values(ascending=False)
)
print(media_per_corso)

# Statistiche multiple per corso
stats_corso: pd.DataFrame = (
    df.groupby("corso")["voto_medio"]
    .agg(["count", "mean", "std", "min", "max"])
    .round(2)
)

groupby avanzato

# Groupby con piu colonne + unstack per tabella pivot
media_corso_anno: pd.DataFrame = (
    df.groupby(["corso", "anno"])["voto_medio"]
    .mean()
    .round(2)
    .unstack()       # anno diventa colonna
)
print(media_corso_anno)
# Filtro + groupby combinati
bravi: pd.DataFrame = df[df["voto_medio"] >= 27]

crediti_regione: pd.Series = (
    bravi.groupby("regione")["crediti"]
    .mean()
    .round(1)
    .sort_values(ascending=False)
)
print(crediti_regione)
Schema tipico: filtrare → raggruppare → aggregare → ordinare
